if(CCT_STRICT_WARNINGS)
    set(CCT_WARNING_OPTIONS -Wall -Wextra -Wpedantic -Werror)
else()
    set(CCT_WARNING_OPTIONS)
endif()

pkg_check_modules(FFTW3 REQUIRED IMPORTED_TARGET fftw3)

    src/fft_roundtrip.cpp

    src/sequence_scan.cpp

    src/causal.cpp

    src/sft.cpp
    src/preference.cpp
    src/knowledge.cpp
    src/inference.cpp
    src/release.cpp
    src/track1.cpp
)
target_include_directories(cct_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(cct_native PUBLIC PkgConfig::FFTW3)
target_compile_options(cct_native PRIVATE ${CCT_WARNING_OPTIONS})
add_executable(cct_gate_envelope tools/gate_envelope.cpp)
target_link_libraries(cct_gate_envelope PRIVATE cct_native)
target_compile_options(cct_gate_envelope PRIVATE ${CCT_WARNING_OPTIONS})
execute_process(COMMAND git -C ${CMAKE_SOURCE_DIR}/.. rev-parse HEAD OUTPUT_VARIABLE CCT_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT CCT_SOURCE_COMMIT)
    set(CCT_SOURCE_COMMIT unknown)
endif()
target_compile_definitions(cct_gate_envelope PRIVATE
    CCT_SOURCE_COMMIT="${CCT_SOURCE_COMMIT}"
    CCT_COMPILER_ID="${CMAKE_CXX_COMPILER_ID}"
    CCT_COMPILER_VERSION="${CMAKE_CXX_COMPILER_VERSION}"
    CCT_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
